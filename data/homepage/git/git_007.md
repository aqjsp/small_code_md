## 4 Git自动化脚本

### 4.1 自动化提交脚本
```bash
#!/bin/bash
# auto-commit.sh

# 检查是否有更改
if [ -z "$(git status --porcelain)" ]; then
    echo "没有更改需要提交"
    exit 0
fi

# 添加所有更改
git add .

# 提交更改
if [ -z "$1" ]; then
    git commit -m "自动提交 $(date)"
else
    git commit -m "$1"
fi

# 推送更改
git push

echo "更改已提交并推送"
```

### 4.2 自动化备份脚本
```bash
#!/bin/bash
# backup-repo.sh

REPO_DIR="/path/to/repo"
BACKUP_DIR="/path/to/backup"
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
BACKUP_NAME="repo_backup_$TIMESTAMP"

# 创建备份目录
mkdir -p "$BACKUP_DIR"

# 克隆仓库
git clone --bare "$REPO_DIR" "$BACKUP_DIR/$BACKUP_NAME"

# 压缩备份
cd "$BACKUP_DIR"
tar -czf "$BACKUP_NAME.tar.gz" "$BACKUP_NAME"
rm -rf "$BACKUP_NAME"

echo "备份完成: $BACKUP_DIR/$BACKUP_NAME.tar.gz"
```

### 4.3 自动化发布脚本
```bash
#!/bin/bash
# release.sh

VERSION=$1
if [ -z "$VERSION" ]; then
    echo "请指定版本号"
    exit 1
fi

# 确保在主分支
git checkout main
git pull origin main

# 创建发布分支
git checkout -b "release/v$VERSION"

# 更新版本号
echo "$VERSION" > VERSION.txt
git add VERSION.txt
git commit -m "更新版本号到 $VERSION"

# 创建标签
git tag -a "v$VERSION" -m "发布版本 $VERSION"

# 推送分支和标签
git push origin "release/v$VERSION"
git push origin "v$VERSION"

echo "发布版本 $VERSION 完成"
```

### 4.4 自动化清理脚本
```bash
#!/bin/bash
# cleanup.sh

# 清理已合并的分支
git branch --merged | grep -v "\*" | grep -v "main" | grep -v "develop" | xargs -n 1 git branch -d

# 清理远程分支
git remote prune origin

# 清理reflog
git reflog expire --expire=now --all

```

### 6.3 Git Hooks
```bash
# 安装pre-commit钩子
#!/bin/sh
# .git/hooks/pre-commit

# 运行代码格式检查
npm run lint
if [ $? -ne 0 ]; then
    echo "代码格式检查失败，请修复后重试"
    exit 1
fi

# 运行测试
npm test
if [ $? -ne 0 ]; then
    echo "测试失败，请修复后重试"
    exit 1
fi

# 检查大文件
MAX_SIZE=$(git config hooks.maxsize 2>/dev/null || echo 10485760)  # 默认10MB
for file in $(git diff --cached --name-only); do
    size=$(git cat-file -s :"$file" 2>/dev/null || echo 0)
    if [ "$size" -gt "$MAX_SIZE" ]; then
        echo "文件 $file 大小超过限制 ($size > $MAX_SIZE)"
        exit 1
    fi
done
```

### 9.3 访问控制
```bash
# 限制推送权限

git config --global receive.denyDeletes true

# 限制分支删除
git config --global receive.denyDeleteCurrent true

# 配置钩子进行访问控制
# .git/hooks/pre-receive
#!/bin/bash
while read oldrev newrev refname; do
    if [[ $refname == refs/heads/main ]]; then
        # 检查推送权限
        if ! has_push_permission $newrev; then
            echo "拒绝推送: 没有main分支的推送权限"
            exit 1
        fi
    fi
done
```

## 10 Git故障排除

### 10.1 常见问题解决
```bash
# 问题1: "fatal: refusing to merge unrelated histories"
# 解决方案:
git pull origin main --allow-unrelated-histories

# 问题2: "error: The following untracked working tree files would be overwritten by checkout"
# 解决方案:
git clean -fd
git checkout branch-name

# 问题3: "fatal: 'origin' does not appear to be a git repository"
# 解决方案:
git remote add origin https://github.com/user/repo.git

# 问题4: "error: failed to push some refs to 'https://github.com/user/repo.git'"
# 解决方案:
git pull --rebase origin main
git push origin main
```

### 10.2 恢复丢失的工作
```bash
# 恢复未提交的更改
git fsck --lost-found
ls .git/lost-found/commit/
git show <commit-hash> > recovered-file.txt

# 恢复删除的分支
git reflog
git checkout -b recovered-branch HEAD@{5}

# 恢复删除的提交
git reflog
git cherry-pick HEAD@{2}

# 恢复重置的提交
git reflog
git reset --hard HEAD@{3}
```

### 10.3 修复损坏的仓库
```bash
# 检查仓库完整性
git fsck --full

# 修复损坏的包文件
git gc --prune=now

# 重建索引
git reset

# 重新打包对象
git repack -a -d --depth=250 --window=250
```
