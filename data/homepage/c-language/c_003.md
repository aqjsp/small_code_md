# C语言控制结构与程序流程控制

## 1. 程序流程控制基础理论

### 1.1 程序执行模型的本质

在深入学习控制结构之前，我们需要理解程序执行的基本模型。程序的执行本质上是指令的序列化执行过程，由CPU的程序计数器(Program Counter, PC)控制，它指向下一条要执行的指令地址。

**程序执行的本质**：
```c
#include <stdio.h>

// 演示程序的基本执行流程
int main() {
    printf("语句1：程序开始执行\n");        // 顺序执行
    printf("语句2：按顺序执行\n");          // 顺序执行
    
    int condition = 1;
    if (condition) {                        // 条件分支
        printf("语句3：条件为真时执行\n");
    }
    
    for (int i = 0; i < 3; i++) {          // 循环重复
        printf("语句4：循环第%d次\n", i + 1);
    }
    
    printf("语句5：程序结束\n");            // 顺序执行
    return 0;
}
```

从计算机体系结构的角度看，程序执行遵循冯·诺依曼模型，即指令和数据存储在同一内存中，CPU按顺序获取、解码和执行指令。控制结构通过修改程序计数器的值来实现流程的改变，这是程序实现逻辑判断和重复操作的基础。

**程序计数器（PC）的作用**：
程序计数器是CPU中的一个特殊寄存器，它存储下一条要执行指令的内存地址。在顺序执行时，PC会自动增加指向下一条指令；当遇到跳转指令时，PC会被设置为新的目标地址。控制结构的本质就是通过条件判断和跳转指令来改变PC的值，从而改变程序的执行流程。

```c
// 汇编层面的理解（伪代码）
// 顺序执行：PC = PC + 指令长度
// 条件跳转：if (condition) PC = target_address else PC = PC + 指令长度
// 无条件跳转：PC = target_address
// 循环：PC = loop_start_address
// 函数调用：push PC; PC = function_address
```

现代处理器通过流水线技术和分支预测来优化控制流的执行效率。分支预测器会猜测条件跳转的结果，并提前获取可能执行的指令，如果预测正确，可以显著提高执行速度；如果预测错误，则需要清空流水线并重新获取正确的指令，这会带来性能损失。

程序的执行模型可以分为三个层次：机器码层面的跳转指令、汇编语言中的分支和循环指令，以及高级语言中的控制结构。高级语言的控制结构通过编译器转换为底层的跳转指令，但为程序员提供了更加抽象和直观的编程模型，使程序员能够专注于逻辑设计而非底层实现细节。

### 1.2 结构化编程的理论基础

结构化编程是20世纪60年代提出的重要编程范式，旨在通过限制控制流程的使用来提高代码的可读性和可维护性。1966年，Corrado Böhm和Giuseppe Jacopini在论文《Flow Diagrams, Turing Machines and Languages with Only Two Formation Rules》中证明了任何可计算函数都可以通过三种基本控制结构来实现：顺序、选择和循环。这一理论被称为Böhm-Jacopini定理，为结构化编程提供了坚实的数学基础。

结构化编程理论的核心思想是通过单入口单出口的控制结构来避免复杂的跳转逻辑，从而降低程序的认知复杂度。与早期广泛使用的goto语句相比，结构化控制结构使代码的执行路径更加清晰，便于理解和维护。Edsger Dijkstra在1968年发表的著名论文《Go To Statement Considered Harmful》中，系统阐述了goto语句对程序质量的危害，推动了结构化编程的普及。

**三种基本控制结构的深入理解**：

```c
#include <stdio.h>

// 1. 顺序结构（Sequence）
void sequence_example() {
    printf("=== 顺序结构演示 ===\n");
    int a = 10;                    // 步骤1
    int b = 20;                    // 步骤2
    int sum = a + b;               // 步骤3
    printf("结果: %d\n", sum);     // 步骤4
    // 每个步骤按照编写顺序依次执行
}

// 2. 选择结构（Selection）
void selection_example(int x) {
    printf("=== 选择结构演示 ===\n");
    if (x > 0) {
        printf("x是正数: %d\n", x);
    } else if (x < 0) {
        printf("x是负数: %d\n", x);
    } else {
        printf("x是零\n");
    }
    // 根据条件选择执行不同的代码分支
}

// 3. 循环结构（Iteration）
void iteration_example() {
    printf("=== 循环结构演示 ===\n");
    for (int i = 1; i <= 5; i++) {
        printf("第%d次循环\n", i);
    }
    // 重复执行代码块直到满足退出条件
}

int main() {
    sequence_example();
    selection_example(10);
    selection_example(-5);
    selection_example(0);
    iteration_example();
    return 0;
}
```

**结构化编程的核心原则**：

结构化编程的核心原则包括单一入口单一出口原则、模块化设计和层次化抽象。单一入口单一出口原则要求每个代码块（如函数、循环、条件分支）只有一个入口点和一个出口点，这使程序的执行流程更加清晰和可预测。模块化设计鼓励将复杂问题分解为独立的、可重用的模块，每个模块负责特定的功能。层次化抽象则允许程序员在不同抽象层次上思考问题，从高层次的算法逻辑到低层次的实现细节。

这些原则共同作用，降低了程序的认知复杂度，使代码更易于理解、测试和维护。结构化编程不仅是一种编码风格，更是一种思维方式，它帮助程序员组织复杂的逻辑，减少错误，提高代码质量。

```c
#include <stdio.h>

// 单一入口单一出口原则
int calculate_grade(int score) {
    // 单一入口：函数只有一个入口点
    
    if (score < 0 || score > 100) {
        return -1;  // 错误情况的出口
    }
    
    if (score >= 90) return 4;      // A
    if (score >= 80) return 3;      // B
    if (score >= 70) return 2;      // C
    if (score >= 60) return 1;      // D
    return 0;                       // F
    
    // 单一出口：虽然有多个return，但逻辑上是单一的出口路径
}

// 避免goto的结构化设计
void structured_vs_goto() {
    printf("=== 结构化编程 vs goto ===\n");
    
    // 不好的做法（使用goto）
    /*
    int i = 0;
    loop_start:
        if (i >= 5) goto loop_end;
        printf("使用goto: %d\n", i);
        i++;
        goto loop_start;
    loop_end:
    */
    
    // 好的做法（结构化）
    for (int i = 0; i < 5; i++) {
        printf("结构化循环: %d\n", i);
    }
}

int main() {
    printf("学生成绩等级: %d\n", calculate_grade(85));
    structured_vs_goto();
    return 0;
}
```

### 1.3 控制结构的性能考量

不同的控制结构在性能上有不同的特点，理解这些特点有助于编写高效的程序。

```c
#include <stdio.h>
#include <time.h>

// 性能测试宏
#define BENCHMARK(name, code) do { \
    clock_t start = clock(); \
    code \
    clock_t end = clock(); \
    double time_spent = ((double)(end - start)) / CLOCKS_PER_SEC; \
    printf("%s 耗时: %.6f秒\n", name, time_spent); \
} while(0)

void performance_comparison() {
    const int ITERATIONS = 10000000;
    volatile int result = 0;  // 防止编译器优化
    
    printf("=== 控制结构性能比较 ===\n");
    
    // 1. if-else链 vs switch
    BENCHMARK("if-else链", {
        for (int i = 0; i < ITERATIONS; i++) {
            int value = i % 10;
            if (value == 0) result += 1;
            else if (value == 1) result += 2;
            else if (value == 2) result += 3;
            else if (value == 3) result += 4;
            else if (value == 4) result += 5;
            else if (value == 5) result += 6;
            else if (value == 6) result += 7;
            else if (value == 7) result += 8;
            else if (value == 8) result += 9;
            else result += 10;
        }
    });
    
    result = 0;  // 重置
    
    BENCHMARK("switch语句", {
        for (int i = 0; i < ITERATIONS; i++) {
            int value = i % 10;
            switch (value) {
                case 0: result += 1; break;
                case 1: result += 2; break;
                case 2: result += 3; break;
                case 3: result += 4; break;
                case 4: result += 5; break;
                case 5: result += 6; break;
                case 6: result += 7; break;
                case 7: result += 8; break;
                case 8: result += 9; break;
                default: result += 10; break;
            }
        }
    });
    
    // 2. 不同循环结构的性能
    result = 0;
    
    BENCHMARK("for循环", {
        for (int i = 0; i < ITERATIONS; i++) {
            result += i;
        }
    });
    
    result = 0;
    
    BENCHMARK("while循环", {
        int i = 0;
        while (i < ITERATIONS) {
            result += i;
            i++;
        }
    });
    
    result = 0;
    
    BENCHMARK("do-while循环", {
        int i = 0;
        do {
            result += i;
            i++;
        } while (i < ITERATIONS);
    });
}

int main() {
    performance_comparison();
    return 0;
}
```

## 2. 条件控制结构详解

### 2.1 布尔逻辑与条件表达式

在C语言中，条件判断基于布尔逻辑，理解真值的定义和逻辑运算是掌握条件控制的基础。布尔逻辑由乔治·布尔在19世纪中叶提出，它定义了一套基于真值（真和假）的代数系统，为现代计算机的二进制逻辑奠定了理论基础。

C语言中的布尔逻辑具有一些独特特点：零值被解释为假，任何非零值都被解释为真。这种设计使得C语言可以灵活地处理各种类型的条件判断，但也要求程序员对类型转换有清晰的认识。C99标准引入了`<stdbool.h>`头文件，提供了`bool`类型以及`true`和`false`常量，使布尔表达式的语义更加明确。

布尔逻辑的核心运算包括与(AND)、或(OR)和非(NOT)，它们分别对应C语言中的`&&`、`||`和`!`运算符。这些运算符遵循严格的优先级和结合性规则，并且具有短路求值特性：对于`&&`运算，如果左操作数为假，则右操作数不会被求值；对于`||`运算，如果左操作数为真，则右操作数不会被求值。短路求值不仅提高了效率，还在某些情况下避免了潜在的运行时错误。

条件表达式的求值过程涉及类型提升、比较运算和逻辑运算三个阶段。首先，比较运算符两边的操作数会进行常规算术转换；然后执行比较操作，结果为0或1；最后，逻辑运算符对比较结果进行组合，得出最终的条件值。理解这一过程有助于编写正确且高效的条件表达式。

```c
#include <stdio.h>
#include <stdbool.h>  // C99标准的布尔类型

void boolean_logic_demo() {
    printf("=== 布尔逻辑演示 ===\n");
    
    // C语言中的真值定义
    printf("真值测试:\n");
    printf("  0 为假: %s\n", 0 ? "真" : "假");
    printf("  1 为真: %s\n", 1 ? "真" : "假");
    printf("  -1 为真: %s\n", -1 ? "真" : "假");
    printf("  100 为真: %s\n", 100 ? "真" : "假");
    
    // 关系运算符
    int a = 10, b = 20;
    printf("\n关系运算 (a=%d, b=%d):\n", a, b);
    printf("  a < b: %d\n", a < b);
    printf("  a > b: %d\n", a > b);
    printf("  a <= b: %d\n", a <= b);
    printf("  a >= b: %d\n", a >= b);
    printf("  a == b: %d\n", a == b);
    printf("  a != b: %d\n", a != b);
    
    // 逻辑运算符
    printf("\n逻辑运算:\n");
    printf("  (a < b) && (b < 30): %d\n", (a < b) && (b < 30));
    printf("  (a > b) || (b > 15): %d\n", (a > b) || (b > 15));
    printf("  !(a == b): %d\n", !(a == b));
    
    // 短路求值演示
    printf("\n短路求值演示:\n");
    int x = 0;
    printf("  x = %d\n", x);
    
    // 逻辑与的短路求值
    if (x != 0 && (100 / x) > 10) {
        printf("  条件为真\n");
    } else {
        printf("  条件为假（避免了除零错误）\n");
    }
    
    // 逻辑或的短路求值
    if (x == 0 || (100 / x) > 10) {
        printf("  条件为真（第一个条件已满足）\n");
    }
}

// 复杂条件表达式的处理
bool is_valid_date(int year, int month, int day) {
    // 分步骤验证，提高可读性
    if (year < 1900 || year > 2100) return false;
    if (month < 1 || month > 12) return false;
    if (day < 1 || day > 31) return false;
    
    // 特殊月份的天数检查
    if (month == 2) {
        // 闰年判断
        bool is_leap = (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
        return day <= (is_leap ? 29 : 28);
    }
    
    // 30天的月份
    if (month == 4 || month == 6 || month == 9 || month == 11) {
        return day <= 30;
    }
    
    return true;  // 31天的月份
}

void test_date_validation() {
    printf("\n=== 日期验证测试 ===\n");
    
    struct {
        int year, month, day;
        const char* description;
    } test_cases[] = {
        {2023, 2, 28, "平年2月28日"},
        {2023, 2, 29, "平年2月29日（无效）"},
        {2024, 2, 29, "闰年2月29日"},
        {2023, 4, 31, "4月31日（无效）"},
        {2023, 12, 31, "12月31日"},
        {1899, 1, 1, "1899年（无效）"}
    };
    
    for (int i = 0; i < 6; i++) {
        bool valid = is_valid_date(test_cases[i].year, 
                                  test_cases[i].month, 
                                  test_cases[i].day);
        printf("  %s: %s\n", test_cases[i].description, 
               valid ? "有效" : "无效");
    }
}

int main() {
    boolean_logic_demo();
    test_date_validation();
    return 0;
}
```

### 2.2 if语句的深入应用

if语句是最基本也是最重要的条件控制结构，掌握其各种形式和应用技巧至关重要。

```c
        printf("恭喜！你及格了，分数：%d\n", score);
```

### 2.3 switch语句的高级应用

switch语句是处理多路分支的高效工具，特别适合处理离散值的选择。

```c
#include <stdio.h>
#include <string.h>

// 1. 基本switch语句
void basic_switch_demo() {
    printf("=== 基本switch语句演示 ===\n");
    
    int day = 3;
    
    switch (day) {

    printf("当前状态: ");
    switch (current_state) {
        case STATE_IDLE: printf("空闲"); break;
```

## 3. 循环控制结构详解

### 3.1 循环的本质与设计原理

循环是程序中实现重复操作的基本机制，它体现了计算思维中的迭代概念，是解决重复性问题的核心工具。从数学角度看，循环实现了递归定义的迭代过程；从计算机科学角度看，循环是控制流的重要组成部分，使程序能够以有限的代码描述无限的重复操作。

循环的设计原理源于图灵机模型中的状态转移机制，通过不断改变程序状态并判断终止条件来实现重复计算。在计算机体系结构中，循环通过程序计数器的跳转机制实现，形成执行路径的闭合回路。现代处理器通过分支预测和循环缓存等技术优化循环执行效率。

循环结构包含四个基本要素：初始化、条件判断、循环体和更新操作。这四个要素的排列组合形成了不同的循环变体，但本质上都遵循相同的执行逻辑。理解循环的本质有助于我们选择合适的循环结构，避免常见的循环陷阱，如无限循环、差一错误等。


### 3.3 while和do-while循环

while循环和do-while循环是C语言中另外两种重要的循环结构，它们与for循环相比具有不同的控制特点和适用场景。这两种循环结构体现了"前置测试"和"后置测试"两种不同的循环控制模式，是结构化编程中处理不确定迭代次数问题的基本工具。

while循环采用前置测试模式，即先判断条件后执行循环体，这种设计源于数学归纳法中的基础情况验证，确保循环在满足条件时才执行。从计算机科学角度看，while循环实现了"当...时重复"的逻辑模式，适合处理循环次数不确定但终止条件明确的场景，如输入验证、搜索算法等。

do-while循环则采用后置测试模式，即先执行循环体后判断条件，这种设计确保循环至少执行一次，体现了"做...直到"的逻辑模式。从实际应用角度看，do-while循环特别适合需要至少执行一次的操作，如菜单系统、输入处理等场景。在程序设计理论中，do-while循环对应于循环不变量的初始验证步骤，是确保算法正确性的重要机制。

这两种循环结构的选择反映了程序设计中的"最小惊讶原则"：当需要确保至少执行一次时选择do-while，当可能零次执行时选择while。理解这两种循环的语义差异和适用场景，是编写高质量、可读性强的代码的基础。

### 3.4 循环控制语句详解

break和continue语句是控制循环执行流程的重要工具，它们提供了对标准循环行为的精确控制能力。这两种控制语句体现了结构化编程中的"早期退出"和"跳过迭代"模式，是处理复杂循环逻辑的基本机制。

break语句实现了循环的早期退出机制，它允许程序在满足特定条件时立即终止当前循环，而不必等待自然循环条件变为假。从控制流理论角度看，break语句提供了一种打破循环不变量的机制，使程序员能够在循环内部检测到特殊情况时提前退出。在计算机体系结构层面，break通过修改程序计数器的值实现非顺序跳转，直接跳转到循环结构后的第一条指令。break语句在搜索算法、错误处理和条件终止等场景中特别有用，能够提高代码效率和可读性。

continue语句则实现了跳过当前迭代剩余部分的机制，它使程序控制流直接跳转到循环的下一次迭代条件检查。从算法设计角度看，continue语句提供了一种过滤不符合条件的数据的优雅方式，避免了深层嵌套的if-else结构。在程序执行模型中，continue通过跳过循环体剩余部分并执行循环更新操作来实现，保持了循环的基本控制结构。continue语句在数据筛选、异常值处理和条件性操作等场景中表现出色，能够简化代码逻辑并提高可维护性。

这两种控制语句的设计体现了编程语言设计中的"表达力与简洁性平衡"原则：它们提供了足够的控制能力，同时避免了goto语句带来的复杂性。合理使用break和continue能够编写出更加清晰、高效的循环代码，但过度使用也可能破坏代码的可读性，因此需要根据具体场景谨慎选择。

## 4. 控制结构的高级应用与最佳实践

控制结构的高级应用体现了程序设计中的组合思维和分层抽象能力，是将基本控制结构组合成复杂算法和系统的基础。从软件工程角度看，控制结构的组合使用反映了"分而治之"的问题解决策略，通过将复杂问题分解为多个简单控制结构的组合，实现代码的模块化和可维护性。

在算法设计理论中，控制结构的组合遵循"结构化组合原则"，即通过顺序、选择和循环三种基本结构的嵌套和组合，可以构建任意复杂的算法流程。这种组合不是简单的堆砌，而是基于问题域的逻辑关系和数据流依赖，形成有机的控制流网络。从认知科学角度看，合理的控制结构组合能够降低代码的认知复杂度，使程序员更容易理解和维护代码。

控制结构的高级应用还体现了"关注点分离"的设计原则，通过不同控制结构处理不同的逻辑层面，如数据验证、业务逻辑处理和结果展示等。这种分层处理方式不仅提高了代码的可读性，也为后续的测试、调试和优化提供了便利。在实际项目中，控制结构的组合使用还需要考虑性能因素，如减少不必要的嵌套、优化循环结构等，以实现代码质量和执行效率的平衡。

### 4.1 控制结构的组合使用

在实际编程中，我们经常需要组合使用多种控制结构来解决复杂问题。控制结构的组合使用体现了程序设计中的"分层处理"思想，通过不同层次的控制结构处理不同复杂度的逻辑问题。从数据流角度看，控制结构的组合形成了数据的处理管道，每个控制结构负责管道中的一个特定处理阶段。

控制结构的组合模式主要有几种典型类型：顺序组合、嵌套组合和并行组合。顺序组合是最基本的组合方式，通过将多个控制结构按顺序排列，形成线性的处理流程；嵌套组合则通过将一个控制结构置于另一个控制结构内部，实现复杂的条件判断和循环控制；并行组合则通过多个独立的控制结构处理不同的数据分支，最后汇总结果。

从软件设计模式角度看，控制结构的组合使用也体现了"策略模式"和"模板方法模式"的思想。策略模式通过不同的控制结构实现不同的算法策略，而模板方法模式则通过固定的控制结构框架定义算法骨架，允许子步骤的变化。合理运用这些组合模式，能够编写出更加灵活、可扩展的代码，同时保持代码的清晰性和可维护性。

### 4.2 性能优化技巧

了解控制结构的性能特点，有助于编写高效的程序。性能优化不仅仅是代码层面的技巧，更是对计算机体系结构、编译器原理和程序执行模型的深入理解。从计算机体系结构角度看，控制结构的性能优化主要围绕减少指令周期、提高缓存命中率和降低分支预测失败率等核心目标。

循环优化是性能优化的重点领域，因为循环通常是程序中的热点代码，占据了大部分执行时间。循环优化的理论基础是"阿姆达尔定律"，即优化程序中最频繁执行的部分能带来最大的性能提升。循环展开、循环不变量外提和循环融合等技术，通过减少循环控制开销、提高指令级并行性和改善数据局部性，显著提升循环执行效率。

条件判断优化则基于分支预测和流水线执行原理。现代处理器采用分支预测技术来猜测条件跳转的结果，预测失败会导致流水线清空，造成性能损失。通过调整条件判断顺序、使用查找表代替复杂条件判断和减少分支数量，可以提高分支预测准确率，降低流水线停顿。

缓存友好的数据访问模式优化源于计算机存储层次结构的特点。CPU缓存、主存和硬盘形成了速度和容量的金字塔结构，缓存未命中会导致数十到数百个时钟周期的延迟。通过优化数据访问模式，提高空间局部性和时间局部性，可以显著减少缓存未命中次数，从而提升程序整体性能。

### 4.3 错误处理与调试技巧

良好的错误处理和调试技巧是编写可靠程序的关键。错误处理机制体现了软件系统对异常情况的响应能力，是构建健壮系统的基础。从软件工程角度看，错误处理不仅仅是代码的技术实现，更是一种系统性的设计思维，需要在系统设计阶段就考虑各种可能的异常情况和应对策略。

错误处理的理论基础源于"防御性编程"理念，即假设程序运行过程中可能出现的各种异常情况，并提前设计相应的处理机制。这种设计思想与"墨菲定律"相呼应——"凡是可能出错的地方，就一定会出错"。在C语言中，错误处理主要通过返回值、错误码和信号机制实现，这些机制形成了C语言错误处理的基本框架。

调试技巧则体现了程序开发的"可观测性"原则，即程序应该提供足够的运行时信息，帮助开发者理解程序状态和定位问题。从认知科学角度看，调试过程本质上是一个假设验证的过程，开发者通过观察程序行为、形成假设、设计实验来验证假设，最终定位问题根源。有效的调试技巧包括断言、日志、条件编译和调试器使用等，这些工具和方法构成了C语言调试的工具箱。

错误处理与调试技巧的结合，形成了C语言程序质量保障的双重防线：错误处理机制在运行时捕获和处理异常，而调试技巧则在开发阶段帮助发现和修复问题。掌握这两方面的技能，是成为优秀C语言程序员的必经之路。

```c
    if (email == NULL || strlen(email) == 0) {

                break;

        }
    }
```

## 5. 实际项目应用案例

实际项目应用案例是理论知识与实践应用的桥梁，它展示了如何将前面学习的控制结构概念转化为解决实际问题的系统设计。从软件架构角度看，任何实际项目都可以视为一个由多个层次组成的系统：表示层（用户界面）、业务逻辑层（核心功能）、数据访问层（数据存储）和基础设施层（系统支持）。控制结构在每一层中都扮演着关键角色，它们构成了系统的"神经系统"，负责协调各组件之间的交互和数据流动。

系统设计的核心原则是"关注点分离"和"模块化"，这意味着将复杂系统分解为相对独立、职责单一的模块，每个模块通过明确的接口与其他模块交互。控制结构是实现这种分离的关键工具：选择结构用于处理不同的业务场景和异常情况，循环结构用于处理批量数据和重复任务，而跳转结构（如break和continue）则用于优化控制流程和提高代码效率。

在实际项目中，控制结构的设计需要考虑多个质量属性，包括性能、可维护性、可扩展性、可靠性和安全性。性能考虑涉及算法效率、资源使用和响应时间；可维护性关注代码结构、命名规范和文档完整性；可扩展性要求系统设计具有灵活性，能够适应需求变化；可靠性需要考虑错误处理、异常恢复和数据一致性；安全性则涉及输入验证、权限控制和数据保护。

本章节通过学生管理系统这一典型案例，展示了如何综合运用各种控制结构构建一个完整的应用程序。这个案例虽然规模不大，但包含了实际项目的主要元素：数据模型设计、业务逻辑实现、用户界面交互、错误处理和系统统计等。通过这个案例，读者可以深入理解控制结构在实际项目中的应用模式，为开发更复杂的系统奠定基础。

### 5.1 简单的学生管理系统

综合运用各种控制结构实现一个完整的应用程序。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_STUDENTS 100
#define MAX_NAME_LENGTH 50

typedef struct {
    int id;
    char name[MAX_NAME_LENGTH];
    int age;
    double scores[3];  // 三门课程成绩
    double average;
} Student;

typedef struct {
    Student students[MAX_STUDENTS];
    int count;
} StudentManager;

// 初始化学生管理器
void init_manager(StudentManager* manager) {
    manager->count = 0;
}

// 计算平均分
double calculate_average(const double scores[3]) {
    return (scores[0] + scores[1] + scores[2]) / 3.0;
}

// 添加学生
bool add_student(StudentManager* manager, int id, const char* name, 
                int age, double score1, double score2, double score3) {
    
    if (manager->count >= MAX_STUDENTS) {
        printf("错误：学生数量已达上限\n");
        return false;
    }
    
    // 检查ID是否已存在
    for (int i = 0; i < manager->count; i++) {
        if (manager->students[i].id == id) {
            printf("错误：学生ID %d 已存在\n", id);
            return false;
        }
    }
    
    // 添加新学生
    Student* new_student = &manager->students[manager->count];
    new_student->id = id;
    strncpy(new_student->name, name, MAX_NAME_LENGTH - 1);
    new_student->name[MAX_NAME_LENGTH - 1] = '\0';
    new_student->age = age;
    new_student->scores[0] = score1;
    new_student->scores[1] = score2;
    new_student->scores[2] = score3;
    new_student->average = calculate_average(new_student->scores);
    
    manager->count++;
    printf("成功添加学生：%s (ID: %d)\n", name, id);
    return true;
}

// 查找学生
Student* find_student_by_id(StudentManager* manager, int id) {
    for (int i = 0; i < manager->count; i++) {
        if (manager->students[i].id == id) {
            return &manager->students[i];
        }
    }
    return NULL;
}

// 删除学生
bool remove_student(StudentManager* manager, int id) {
    for (int i = 0; i < manager->count; i++) {
        if (manager->students[i].id == id) {
            // 将后面的学生前移
            for (int j = i; j < manager->count - 1; j++) {
                manager->students[j] = manager->students[j + 1];
            }
            manager->count--;
            printf("成功删除学生 ID: %d\n", id);
            return true;
        }
    }
    printf("错误：未找到学生 ID: %d\n", id);
    return false;
}

// 显示单个学生信息
void display_student(const Student* student) {
    printf("ID: %d, 姓名: %s, 年龄: %d\n", 
           student->id, student->name, student->age);
    printf("成绩: %.1f, %.1f, %.1f, 平均分: %.2f\n",
           student->scores[0], student->scores[1], 
           student->scores[2], student->average);
}

// 显示所有学生
void display_all_students(const StudentManager* manager) {
    if (manager->count == 0) {
        printf("暂无学生记录\n");
        return;
    }
    
    printf("\n=== 所有学生信息 ===\n");
    for (int i = 0; i < manager->count; i++) {
        printf("学生 %d:\n", i + 1);
        display_student(&manager->students[i]);
        printf("\n");
    }
}

// 按平均分排序
void sort_by_average(StudentManager* manager) {
    // 使用冒泡排序
    for (int i = 0; i < manager->count - 1; i++) {
        for (int j = 0; j < manager->count - i - 1; j++) {
            if (manager->students[j].average < manager->students[j + 1].average) {
                // 交换学生记录
                Student temp = manager->students[j];
                manager->students[j] = manager->students[j + 1];
                manager->students[j + 1] = temp;
            }
        }
    }
    printf("学生记录已按平均分降序排列\n");
}

// 统计信息
void show_statistics(const StudentManager* manager) {
    if (manager->count == 0) {
        printf("暂无学生记录，无法统计\n");
        return;
    }
    
    printf("\n=== 统计信息 ===\n");
    printf("学生总数: %d\n", manager->count);
    
    // 计算各科平均分和总平均分
    double subject_totals[3] = {0, 0, 0};
    double total_average = 0;
    int excellent_count = 0;  // 平均分>=90
    int good_count = 0;       // 平均分>=80
    int pass_count = 0;       // 平均分>=60
    
    for (int i = 0; i < manager->count; i++) {
        const Student* s = &manager->students[i];
        
        for (int j = 0; j < 3; j++) {
            subject_totals[j] += s->scores[j];
        }
        
        total_average += s->average;
        
        if (s->average >= 90) {
            excellent_count++;
        } else if (s->average >= 80) {
            good_count++;
        } else if (s->average >= 60) {
            pass_count++;
        }
    }
    
    printf("各科平均分:\n");
    printf("  科目1: %.2f\n", subject_totals[0] / manager->count);
    printf("  科目2: %.2f\n", subject_totals[1] / manager->count);
    printf("  科目3: %.2f\n", subject_totals[2] / manager->count);
    printf("总体平均分: %.2f\n", total_average / manager->count);
    
    printf("成绩分布:\n");
    printf("  优秀 (>=90): %d人 (%.1f%%)\n", 
           excellent_count, (double)excellent_count / manager->count * 100);
    printf("  良好 (80-89): %d人 (%.1f%%)\n", 
           good_count, (double)good_count / manager->count * 100);
    printf("  及格 (60-79): %d人 (%.1f%%)\n", 
           pass_count, (double)pass_count / manager->count * 100);
    printf("  不及格 (<60): %d人 (%.1f%%)\n", 
           manager->count - excellent_count - good_count - pass_count,
           (double)(manager->count - excellent_count - good_count - pass_count) / manager->count * 100);
}

// 主菜单
void show_menu() {

    printf("1. 添加学生\n");
    printf("2. 查找学生\n");
    printf("3. 删除学生\n");
```
